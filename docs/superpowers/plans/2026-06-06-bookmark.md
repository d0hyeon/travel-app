```ts
// src/features/place/useBookmark.ts
import { useEffect, useState } from 'react'
import { clientDatabase } from '~app/client-database'

export function useBookmark(placeId: string) {
  const [isBookmarked, setIsBookmarked] = useState(false)

  useEffect(() => {
    clientDatabase.bookmarks.get(placeId).then((item) => {
      setIsBookmarked(item != null)
    })
  }, [placeId])

  const toggle = async () => {
    if (isBookmarked) {
      await clientDatabase.bookmarks.delete(placeId)
      setIsBookmarked(false)
    } else {
      await clientDatabase.bookmarks.add({ placeId, savedAt: Date.now() })
      setIsBookmarked(true)
    }
  }

  return { isBookmarked, toggle }
}
```

```ts
// src/features/place/useBookmarks.ts
import { useEffect, useState } from 'react'
import { clientDatabase } from '~app/client-database'

export function useBookmarks() {
  const [bookmarks, setBookmarks] = useState<{ placeId: string; savedAt: number }[]>([])

  useEffect(() => {
    clientDatabase.bookmarks.getAll().then(setBookmarks)
  }, [])

  return { bookmarks }
}
```
